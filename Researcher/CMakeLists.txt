# Separate research application. Reuse archive chart cards and the domain formulas.
set(RESEARCH_SOURCES ResearchWindow.cpp ResearchWindow.h ResearchStore.cpp ResearchStore.h
    ../Labeler/ChartFrame.cpp ../Labeler/ChartFrame.h
    ../Labeler/SessionScanner.cpp ../Labeler/SessionScanner.h
    ../Viewer/PressureAnalysis.cpp ../Viewer/PressureAnalysis.h)
qt_add_executable(DataResearcher main.cpp ${RESEARCH_SOURCES}
    ${CMAKE_SOURCE_DIR}/img.qrc ${CMAKE_SOURCE_DIR}/fonts.qrc ${CMAKE_SOURCE_DIR}/style.qrc)
target_include_directories(DataResearcher PRIVATE ../Labeler ../Viewer)
target_link_libraries(DataResearcher PRIVATE Gui Qt6::Widgets)
target_compile_definitions(DataResearcher PRIVATE MPIPLATFORM_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
set_target_properties(DataResearcher PROPERTIES WIN32_EXECUTABLE TRUE BUILD_WITH_INSTALL_RPATH FALSE)
if(NOT MPIPLATFORM_EXPERIMENTAL_LINEARITY)
    target_sources(DataResearcher PRIVATE ${CMAKE_SOURCE_DIR}/Src/Domain/Tests/Main/Linearity/LinearityMethods.cpp)
endif()
if(MPIPLATFORM_MINGW)
    target_link_options(DataResearcher PRIVATE -static-libgcc -static-libstdc++ -static -Wl,-subsystem,windows)
endif()
get_target_property(_research_qt_type Qt6::Core TYPE)
if(_research_qt_type STREQUAL "STATIC_LIBRARY")
    include("${Qt6Gui_DIR}/Qt6QSvgPluginConfig.cmake")
    include("${Qt6Gui_DIR}/Qt6QSvgIconPluginConfig.cmake")
    include("${Qt6Gui_DIR}/Qt6QWindowsIntegrationPluginConfig.cmake")
    target_link_libraries(DataResearcher PRIVATE Qt6::QSvgPlugin Qt6::QSvgIconPlugin Qt6::QWindowsIntegrationPlugin)
    target_compile_definitions(DataResearcher PRIVATE MPIPLATFORM_STATIC_QT_PLUGINS)
endif()
if(MPIPLATFORM_BUILD_TESTS)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(ResearcherTests ResearcherTest.cpp ${RESEARCH_SOURCES})
    target_include_directories(ResearcherTests PRIVATE ../Labeler ../Viewer)
    target_link_libraries(ResearcherTests PRIVATE Gui Qt6::Test)
    set_target_properties(ResearcherTests PROPERTIES BUILD_WITH_INSTALL_RPATH FALSE)
    if(NOT MPIPLATFORM_EXPERIMENTAL_LINEARITY)
        target_sources(ResearcherTests PRIVATE ${CMAKE_SOURCE_DIR}/Src/Domain/Tests/Main/Linearity/LinearityMethods.cpp)
    endif()
    target_compile_definitions(ResearcherTests PRIVATE RESEARCH_FIXTURE_ROOT="${CMAKE_SOURCE_DIR}/research/local-slope-stability")
    add_test(NAME ResearcherTests COMMAND ResearcherTests)
    set_tests_properties(ResearcherTests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
